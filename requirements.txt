# UNICA FUENTE DE VERDAD DE DEPENDENCIAS.
#
# El Dockerfile ya no instala nada por fuera de este archivo. Hasta la version
# anterior habia un `pip install langgraph chromadb scikit-learn` suelto, sin
# version, encima de un requirements.txt que declaraba la politica contraria.
# Dos reconstrucciones de la imagen en semanas distintas producian sistemas
# distintos, y para un proceso que coloca ordenes eso convierte cualquier
# diagnostico en una conjetura: no se puede afirmar que produccion corre lo
# mismo que se probo.
#
# Todas las versiones de abajo fueron verificadas contra PyPI el 21/08/2026.
# Para actualizar: leer el changelog, cambiar el numero aca, correr la suite
# completa, y recien entonces promover.
# ============================================================================

# --- Broker de ejecucion ----------------------------------------------------
# VERIFICADO: el paquete se llama `ppi-client` (el artefacto instalado es
# ppi_client). La ultima version publicada es 1.2.4 (21/05/2025).
# Hasta la version anterior este archivo fijaba `ppi-client==0.3.2`, un numero
# puesto sin poder consultar PyPI. Esa version no es la vigente: el build
# habria fallado, o traido una API distinta a la que espera c_ppi_client.py.
#
# ADVERTENCIA DE MANTENIMIENTO: el paquete no publica releases desde hace mas
# de doce meses. ae_ppi_api_watch.py vigila PyPI y avisa por Telegram si
# aparece una version nueva; si quedara definitivamente abandonado, el plan es
# forkear y fijar el fork, no aflojar el pin.
ppi-client==1.2.4

# --- WebSocket del broker --------------------------------------------------
# ppi-client 1.2.4 depende de signalrcorePPI 0.9.2, que exige exactamente
# websocket-client 1.0.0. Se fija ese valor para que el entorno sea resoluble.
websocket-client==1.0.0

# --- Futuros: Matba Rofex / reMarkets -------------------------------------
# pyRofex 0.5.0 exige websocket-client>=1.6.4 y por lo tanto NO puede convivir
# en el mismo entorno Python con el cliente oficial de PPI. Mantener ambos en
# requirements.txt hacia imposible el build. En esta entrega Sandbox queda
# ROFEX_ENABLED=false y av_rofex_client.py degrada de forma cerrada si la
# libreria no esta instalada. Para habilitar futuros se requiere aislar
# pyRofex en un servicio separado; no se debe forzar pip ni romper PPI.

# --- Motor de decision (IA) -------------------------------------------------
# SDK oficial vigente de Google para Gemini. El paquete anterior
# `google-generativeai` esta DEPRECADO (fin de vida 30/11/2025) y no soporta
# la serie Gemini 3. La version minima 1.51.0 es la que habilita las features
# de Gemini 3, incluido `thinking_level`.
google-genai>=1.51.0

# Motor de respaldo. Deja de ser opcional-comentado: si el respaldo no se
# instala, el sistema no tiene a donde caer cuando Gemini falla, y el
# resultado es que deja de abrir posiciones. La llave sigue siendo opcional
# (ANTHROPIC_API_KEY vacia = respaldo desactivado); lo que no debe ser
# opcional es la libreria.
anthropic>=1.0.0

# --- Datos de mercado y macro ----------------------------------------------
yfinance>=0.2.40
feedparser>=6.0.0
requests>=2.32.0

# --- Calculo numerico e indicadores tecnicos -------------------------------
# pandas 3.x introduce cambios de API respecto de la serie 2.x. Se fija la
# serie 2.x: migrar a 3.x es un trabajo con tests de regresion propios, no un
# cambio de numero en este archivo.
pandas>=2.2.0,<3.0.0
numpy>=1.26.0,<3.0.0

# TA-Lib requiere compilar una libreria C: en un contenedor slim es una fuente
# recurrente de builds rotos y de imagenes de 900 MB. Se usa la alternativa
# pura de Python, que cubre los indicadores que el sistema realmente calcula
# (RSI, MACD, ATR, medias moviles).
ta>=0.11.0

# --- Panel web --------------------------------------------------------------
fastapi>=0.115.0
uvicorn>=0.30.0
python-multipart>=0.0.9
python-dotenv>=1.0.0
apscheduler>=3.10.0

# --- Arquitectura multi-agente, motor SRE (RAG) y AIOps --------------------
# Fijadas: estas tres eran las que el Dockerfile instalaba sueltas y sin
# version. chromadb se fija en la MISMA version que la imagen del servicio de
# compose, porque un desajuste entre cliente y servidor de un indice vectorial
# se manifiesta como resultados vacios, no como un error.
langgraph==1.2.11
chromadb==0.5.23
