def read_bytes(path):
    fd = os.open(path, os.O_RDONLY | os.O_NOFOLLOW | os.O_NOATIME)
    with os.fdopen(fd, 'rb') as stream:
        return stream.read()

def snapshot(path):
    info = path.lstat()
    details = {key: getattr(info, key) for key in (
        'st_dev', 'st_ino', 'st_nlink', 'st_uid', 'st_gid', 'st_mode',
        'st_size', 'st_atime_ns', 'st_mtime_ns', 'st_ctime_ns')}
    if stat.S_ISREG(info.st_mode):
        fd = os.open(path, os.O_RDONLY | os.O_NOFOLLOW | os.O_NOATIME)
        with os.fdopen(fd, 'rb') as stream:
            digest = hashlib.sha256()
            while data := stream.read(1024 * 1024): digest.update(data)
        details['sha256'] = digest.hexdigest()
    return details

def snapshot_tree(path):
    result = {str(path): snapshot(path)}
    fd = os.open(path, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW | os.O_NOATIME)
    try:
        with os.scandir(fd) as entries:
            for entry in entries:
                child = path / entry.name
                if entry.is_dir(follow_symlinks=False): result.update(snapshot_tree(child))
                else: result[str(child)] = snapshot(child)
    finally:
        os.close(fd)
    return result

# A full canonical archive is checked against native committed blobs and modes.
entries = subprocess.check_output(['git', '-C', '/workspace/porota_rc6_convergence',
    'ls-tree', '-rz', '--full-tree', SHA]).split(b'\0')
archive_count = 0
for entry in entries:
    if not entry: continue
    description, name = entry.split(b'\t', 1)
    mode, kind, blob = description.decode().split()
    path = REPO / name.decode()
    assert kind == 'blob' and mode in {'100644', '100755'}
    raw = read_bytes(path)
    assert hashlib.sha1(b'blob ' + str(len(raw)).encode() + b'\0' + raw).hexdigest() == blob
    assert not path.is_symlink() and path.stat().st_nlink == 1
    assert stat.S_IMODE(path.stat().st_mode) == (0o755 if mode == '100755' else 0o644)
    archive_count += 1
